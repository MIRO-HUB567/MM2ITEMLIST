local function trim(s) return s:match("^%s*(.-)%s*$") end

local function parseValue(d) local v=d:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>") return v and tonumber(v:gsub(",","")) or nil end
local function extractItems(h) local t={} for n,d in h:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do n=n:match("[^<]+") if n then n=trim(n:gsub("%s+"," ")) n=trim((n:split(" Click "))[1]):lower() local v=parseValue(d) if v then t[n]=v end end end return t end
local function extractChroma(h) local t={} for n,d in h:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do n=n:match("[^<]+") if n then n=trim(n:gsub("%s+"," ")):lower() local v=parseValue(d) if v then t[n]=v end end end return t end
